# Hyprland desktop (official repos). Installed by the "desktop" stage.
# GPU drivers are selected separately from packages/gpu-*.txt.

# Login greeter on tty1 (configured by the services stage from etc/greetd/config.toml)
greetd greetd-tuigreet

# Compositor and Hypr ecosystem (hyprshutdown is bound to SUPER+M)
hyprland hyprpaper hypridle hyprlock hyprpicker hyprshot hyprsunset hyprshutdown
xdg-desktop-portal-hyprland xdg-desktop-portal-gtk

# Polkit agent (launched from hyprland.conf)
polkit-gnome

# Bar, notifications (mako stow group), on-screen display for media keys
waybar mako swayosd

# Tools referenced by hyprland.conf / hypridle.conf / waybar
# (swayosd brightness control needs the user in the "video" group; the services stage adds it)
brightnessctl playerctl pamixer wiremix libnotify libqalculate

# Qt/Wayland integration and the file manager bound to SUPER+E. Outside Plasma,
# dolphin needs kde-cli-tools for "Open With" and the thumbnailers for previews.
qt5-wayland qt6-wayland dolphin kde-cli-tools kdegraphics-thumbnailers ffmpegthumbs ark

# Removable-drive auto-mounting (udiskie is started from hyprland.conf)
udiskie

# Secret storage for browsers/git/VS Code; unlocked at login via PAM (services stage).
# gcr-4 (dependency) provides the gcr-ssh-agent user socket used by .bashrc.
gnome-keyring seahorse


# Bluetooth stack (Waybar bluetooth module; blueberry GUI comes from the AUR)
bluez bluez-utils

# Audio session (installed by archinstall's PipeWire profile; listed so a
# minimal profile still works)
pipewire pipewire-pulse wireplumber

# Fonts used by the configs:
#   ghostty/hyprlock/mako -> "CaskaydiaMono Nerd Font" (ttf-cascadia-mono-nerd)
#   waybar             -> "JetbrainsMono Nerd Font"  (ttf-jetbrains-mono-nerd)
#   waybar icons are Nerd Font glyphs (Material Design / Font Awesome ranges)
ttf-cascadia-mono-nerd ttf-jetbrains-mono-nerd
noto-fonts noto-fonts-emoji noto-fonts-cjk noto-fonts-extra

# Desktop integration
xdg-utils xdg-user-dirs

# Applications (official repos; the rest come from the AUR)
# tailscale: tailscaled.service is enabled by the services stage, then run "sudo tailscale up"
# spotify-launcher: downloads the Spotify client; flags in the spotify stow group
obsidian spotify-launcher tailscale
